# calculate quarterly GDP growth for a sector
# assumes gdp exists and has sector, quarter, and amount columns
# accepts a sector name or a sector factor index, with optional year limits
getGrowth <- function(sector_name, startYear = NULL, endYear = NULL) {
  # check that the data is available
  if (!exists("gdp", envir = .GlobalEnv)) {
    stop("The global 'gdp' data frame does not exist.")
  }

  # check and resolve sector name or index
  if (is.character(sector_name) && length(sector_name) == 1 && !is.na(sector_name)) {
    if (!sector_name %in% levels(gdp$sector)) {
      stop("The specified sector does not exist.")
    }
    target_sector <- sector_name
  } else if (is.numeric(sector_name) && length(sector_name) == 1 &&
             is.finite(sector_name) && sector_name == floor(sector_name) &&
             sector_name >= 1 && sector_name <= length(levels(gdp$sector))) {
    target_sector <- levels(gdp$sector)[sector_name]
  } else {
    stop("sector_name must be a valid sector name or factor index.")
  }

  # check optional year values
  if (!is.null(startYear) && (!is.numeric(startYear) || length(startYear) != 1 ||
                              !is.finite(startYear))) {
    stop("startYear must be a single finite number.")
  }
  if (!is.null(endYear) && (!is.numeric(endYear) || length(endYear) != 1 ||
                            !is.finite(endYear))) {
    stop("endYear must be a single finite number.")
  }
  if (!is.null(startYear) && !is.null(endYear) && startYear > endYear) {
    stop("startYear must not be later than endYear.")
  }

  # extract and sort sector data
  sector_subset <- gdp[gdp$sector == target_sector, ]
  sector_subset <- sector_subset[order(sector_subset$quarter), ]
  sector_amount <- sector_subset$amount
  sector_quarter <- sector_subset$quarter

  # calculate quarterly growth rates
  growth_rates <- diff(sector_amount) / head(sector_amount, -1) * 100
  aligned_quarters <- tail(sector_quarter, -1)

  # store quarters and growth rates
  result_df <- data.frame(
    quarter = aligned_quarters,
    rate = growth_rates
  )

  # filter results by year
  quarter_years <- as.numeric(substr(as.character(result_df$quarter), 1, 4))
  if (!is.null(startYear)) {
    result_df <- result_df[quarter_years >= startYear, ]
    quarter_years <- quarter_years[quarter_years >= startYear]
  }
  if (!is.null(endYear)) {
    quarter_years <- as.numeric(substr(as.character(result_df$quarter), 1, 4))
    result_df <- result_df[quarter_years <= endYear, ]
  }

  rownames(result_df) <- NULL
  return(result_df)
}

# check the GDP data format
isGDPData <- function(df) {
  if (!is.data.frame(df)) {
    stop("Input must be a data frame.")
  }

  if (ncol(df) != 3 || !all(c("sector", "quarter", "amount") %in% names(df))) {
    return(FALSE)
  }
  if (!is.factor(df$sector)) return(FALSE)
  if (!is.factor(df$quarter) && !is.character(df$quarter)) return(FALSE)
  if (!is.numeric(df$amount) || anyNA(df$amount) || any(!is.finite(df$amount))) {
    return(FALSE)
  }
  if (anyNA(df$sector) || anyNA(df$quarter) || nrow(df) == 0) return(FALSE)

  # check that quarters use the expected format
  valid_quarters <- grepl("^[0-9]{4}Q[1-4]$", as.character(df$quarter))
  if (!all(valid_quarters)) return(FALSE)

  return(TRUE)
}
