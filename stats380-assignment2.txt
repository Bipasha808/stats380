# calculate quarterly GDP growth without using gdp directly in the function body
getGrowth <- function(sector_name, startYear = NULL, endYear = NULL, df = gdp) {
  # check data format
  if (!isGDPData(df)) {
    stop("The provided data frame does not match the expected GDP data structure.")
  }

  # check and resolve sector name or index
  if (is.character(sector_name) && length(sector_name) == 1 && !is.na(sector_name)) {
    if (!sector_name %in% levels(df$sector)) {
      stop("The specified sector does not exist.")
    }
    target_sector <- sector_name
  } else if (is.numeric(sector_name) && length(sector_name) == 1 &&
             is.finite(sector_name) && sector_name == floor(sector_name) &&
             sector_name >= 1 && sector_name <= length(levels(df$sector))) {
    target_sector <- levels(df$sector)[sector_name]
  } else {
    stop("sector_name must be a valid sector name or factor index.")
  }

  # check optional year values
  if (!is.null(startYear) && (!is.numeric(startYear) || length(startYear) != 1 ||
                              !is.finite(startYear))) {
    stop("startYear must be a single finite number.")
  }
  if (!is.null(endYear) && (!is.numeric(endYear) || length(endYear) != 1 ||
                            !is.finite(endYear))) {
    stop("endYear must be a single finite number.")
  }
  if (!is.null(startYear) && !is.null(endYear) && startYear > endYear) {
    stop("startYear must not be later than endYear.")
  }

  # extract and sort sector data
  sector_subset <- df[df$sector == target_sector, ]
  sector_subset <- sector_subset[order(sector_subset$quarter), ]
  sector_amount <- sector_subset$amount
  sector_quarter <- sector_subset$quarter

  # calculate quarterly growth rates
  growth_rates <- diff(sector_amount) / head(sector_amount, -1) * 100
  aligned_quarters <- tail(sector_quarter, -1)

  # store quarters and growth rates
  result_df <- data.frame(
    quarter = aligned_quarters,
    rate = growth_rates
  )

  # filter results by year
  quarter_years <- as.numeric(substr(as.character(result_df$quarter), 1, 4))
  if (!is.null(startYear)) {
    result_df <- result_df[quarter_years >= startYear, ]
    quarter_years <- quarter_years[quarter_years >= startYear]
  }
  if (!is.null(endYear)) {
    quarter_years <- as.numeric(substr(as.character(result_df$quarter), 1, 4))
    result_df <- result_df[quarter_years <= endYear, ]
  }

  rownames(result_df) <- NULL
  return(result_df)
}

# check the GDP data format
isGDPData <- function(df) {
  if (!is.data.frame(df)) {
    stop("Input must be a data frame.")
  }

  if (ncol(df) != 3 || !all(c("sector", "quarter", "amount") %in% names(df))) {
    return(FALSE)
  }
  if (!is.factor(df$sector)) return(FALSE)
  if (!is.factor(df$quarter) && !is.character(df$quarter)) return(FALSE)
  if (!is.numeric(df$amount) || anyNA(df$amount) || any(!is.finite(df$amount))) {
    return(FALSE)
  }
  if (anyNA(df$sector) || anyNA(df$quarter) || nrow(df) == 0) return(FALSE)

  # check that quarters use the expected format
  valid_quarters <- grepl("^[0-9]{4}Q[1-4]$", as.character(df$quarter))
  if (!all(valid_quarters)) return(FALSE)

  return(TRUE)
}

# calculate annual growth from rolling four-quarter GDP totals
getAnnualGrowth <- function(sector_name, startYear = NULL, endYear = NULL, df = gdp) {
  # check data format
  if (!isGDPData(df)) {
    stop("The provided data frame does not match the expected GDP data structure.")
  }

  # check and resolve sector name or index
  if (is.character(sector_name) && length(sector_name) == 1 && !is.na(sector_name)) {
    if (!sector_name %in% levels(df$sector)) {
      stop("The specified sector does not exist.")
    }
    target_sector <- sector_name
  } else if (is.numeric(sector_name) && length(sector_name) == 1 &&
             is.finite(sector_name) && sector_name == floor(sector_name) &&
             sector_name >= 1 && sector_name <= length(levels(df$sector))) {
    target_sector <- levels(df$sector)[sector_name]
  } else {
    stop("sector_name must be a valid sector name or factor index.")
  }

  # extract and sort sector data
  sector_subset <- df[df$sector == target_sector, ]
  sector_subset <- sector_subset[order(sector_subset$quarter), ]
  amounts <- sector_subset$amount
  quarters <- sector_subset$quarter
  n <- length(amounts)

  # check that there are enough quarters
  if (n < 4) {
    stop("There are not enough quarterly data points to calculate annual growth.")
  }

  # calculate rolling four-quarter GDP totals
  rolling_sums <- amounts[4:n] + amounts[3:(n - 1)] +
    amounts[2:(n - 2)] + amounts[1:(n - 3)]

  # calculate growth between consecutive rolling totals
  annual_rates <- diff(rolling_sums) / head(rolling_sums, -1) * 100
  padded_quarters <- quarters[2:n]
  padded_annual_rates <- c(rep(NA, 3), annual_rates)

  # store quarters and annual growth rates
  result_df <- data.frame(
    quarter = padded_quarters,
    annualrate = padded_annual_rates
  )

  # filter results by year
  quarter_years <- as.numeric(substr(as.character(result_df$quarter), 1, 4))
  if (!is.null(startYear)) {
    result_df <- result_df[quarter_years >= startYear, ]
    quarter_years <- quarter_years[quarter_years >= startYear]
  }
  if (!is.null(endYear)) {
    quarter_years <- as.numeric(substr(as.character(result_df$quarter), 1, 4))
    result_df <- result_df[quarter_years <= endYear, ]
  }

  rownames(result_df) <- NULL
  return(result_df)
}

# combine quarterly and annual growth rates
getGrowths <- function(sector_name, startYear = NULL, endYear = NULL, df = gdp) {
  # calculate quarterly and annual growth rates
  q_df <- getGrowth(df = df, sector_name = sector_name,
                    startYear = startYear, endYear = endYear)
  a_df <- getAnnualGrowth(df = df, sector_name = sector_name,
                          startYear = startYear, endYear = endYear)

  # match annual rates to the quarterly results
  annual_rates <- a_df$annualrate[match(q_df$quarter, a_df$quarter)]
  result_df <- data.frame(
    quarter = q_df$quarter,
    quarterlyRate = q_df$rate,
    annualRate = annual_rates
  )

  rownames(result_df) <- NULL
  return(result_df)
}

# Plot quarterly and annual growth
plotGrowths <- function(sector_name, startYear = NULL, endYear = NULL,
                        df = gdp, col = c("lightblue", "red")) {
  growths <- getGrowths(sector_name, startYear, endYear, df)

  if (length(col) != 2) {
    stop("col must contain two colours")
  }

  n <- nrow(growths)
  x <- seq_len(n)

  # Find the first quarter of each year
  years <- as.integer(substr(as.character(growths$quarter), 1, 4))
  year_positions <- which(!duplicated(years))

  # Set up the plot
  plot.new()
  plot.window(xlim = c(0.5, n + 0.5), ylim = c(-20, 20))

  # Draw horizontal reference lines
  for (y in seq(-20, 20, by = 5)) {
    abline(h = y, col = "grey80", lty = 2)
  }

  # Draw quarterly growth bars
  rect(x - 0.4, 0, x + 0.4, growths$quarterlyRate,
       col = col[1], border = col[1])

  # Draw annual growth curve
  lines(x, growths$annualRate, col = col[2], lwd = 2)

  # Add axes and labels
  axis(1, at = year_positions, labels = years[year_positions])
  axis(2)
  box()
  title(main = paste("GDP Growth:", sector_name),
        xlab = "Year", ylab = "Growth (%)")

  # Add legend
  legend("topright",
         legend = c("Quarterly growth", "Annual growth"),
         fill = c(col[1], NA),
         border = c(col[1], NA),
         lty = c(NA, 1),
         col = c(col[1], col[2]),
         lwd = c(NA, 2),
         bty = "n")
}

# Plot multiple sectors
plotSectors <- function(sectors, startYear = NULL, endYear = NULL,
                        df = gdp) {

  # Check the number of sectors
  if (length(sectors) == 0) {
    stop("At least one sector must be provided")
  }

  # Set up the plot layout
  old_par <- par(no.readonly = TRUE)
  on.exit(par(old_par))

  n <- length(sectors)
  par(mfrow = c(ceiling(n / 2), 2))

  # Plot each sector
  for (sector_name in sectors) {
    plotGrowths(sector_name, startYear, endYear, df)
  }
}
